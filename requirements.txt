# ============================================================================
# Требования агента. Две группы, и разница между ними не в удобстве, а в
# владельце пакета.
#
#   1. РАНТИМ АГЕНТА   — то, что запускает и обслуживает сам агент.
#   2. ТРАНСПОРТ К enterprise-mcp — то, что нужно агенту, чтобы поговорить
#      с сервером платформы. Сервер поднимается отдельным процессом, и его
#      собственные зависимости здесь не перечислены принципиально.
#
# Вторую группу отделяет не мода, а приёмка фазы 10: «чистое окружение с
# nanobot-ai==0.3.5 поднимается и работает, сервер не требует Nanobot,
# платформа обновляется независимо от агента». Пока в этом файле лежит
# вперемешку с нуждой сервера, обновление платформы тянет за собой pip
# install в окружение агента и наоборот.
#
# Пакеты данных, индексов и эмбеддингов (duckdb, faiss-cpu, numpy, pyarrow,
# sqlglot) и клиент LLM здесь отсутствуют намеренно (п. 10.3, снято
# 2026-10-01 и 2026-10-02): снимком, FAISS-индексами, эмбеддингами и
# проверкой SQL владеют capability data / vectors / llm платформы, и каждый
# пакет объявлен в её манифестах — mcp-platform/requirements.txt и
# mcp-platform/pyproject.toml. Перед удалением проверено разбором AST по
# 509 файлам агента (lib, tools, workspace, tests, contracts) — ни один
# исполняемый модуль их не импортирует.
#
# Сейчас здесь 0 необъявленных импортов и 0 мёртвых пакетов: проверено
# тем же разбором AST по 480 .py агента (lib, workspace, tools, tests,
# contracts, benchmarks) 2026-10-02. Правка, которая вводит в этот файл
# лишний пакет или роняет нужный, ломает 3320 тестов агента.
# ============================================================================


# ----------------------------------------------------------------------------
# 1. РАНТИМ АГЕНТА
#
# Правило: сюда попадает пакет, который импортирует код агента (lib/,
# tools/, workspace/) либо который требуется самому nanobot-ai. Пакет,
# нужный платформе, сюда не попадает, даже если агент его импортирует
# транзитивно, — иначе группы снова сольются.
# ----------------------------------------------------------------------------

# Каркас агента: цикл, каналы, tools, сессии, LLM-провайдеры. Обязателен:
# без него не запускается ни gateway, ни CLI-агент.
nanobot-ai==0.3.5

# Журналирование. Импортируется 21 модулем агента (lib/services, lib/hooks,
# lib/core, gateway.py) — это не выбор, а способ логирования наверху.
loguru==0.7.3

# Канал PostgreSQL: lib/channels/postgres_channel.py. Плюс пул записи
# журнала и startup-проверка схемы в lib/services/db_logging_service.py и
# lib/services/schema_validation.py. psycopg2 нужен агенту самому: к
# PostgreSQL он ходит напрямую, а не через сервер платформы.
psycopg2-binary==2.9.12

# Канал Redis: lib/channels/redis_channel.py. Второй канал агента; на
# платформе Redis не используется вовсе, поэтому в группе 2 его нет.
redis==8.0.0

# Требуется 31 модулем самого nanobot-ai (включая провайдер LLM). Код агента
# не импортирует httpx напрямую, но после сноса lib/services/llm_client.py
# пакет остался нужен каркасу, поэтому версия зафиксирована здесь, а не
# отдана на волю резолвера.
httpx==0.28.1

# Требуется nanobot-ai для чтения конфигурации (config.json, project.json).
# Прямых импортов в коде агента нет — это транзитивная зависимость каркаса,
# зафиксированная явно, чтобы её версия не зависела от резолвера.
PyYAML==6.0.3


# --- Офисные документы ---------------------------------------------------
# Разбор DOCX/XLSX/XLS/PPTX/PDF. Импортирует код агента: навык
# legal_summarizer (workspace/skills/_legal_summarizer/scripts/document/) и
# его тесты. Сам парсер при этом живёт в платформе
# (mcp-platform/libs/office/parser.py) — то есть пакеты нужны обеим
# сторонам, и это общий пересечный набор, а не группа enterprise-mcp:
# агент запускает его в своих тестах (tests/test_office_files.py,
# tests/test_structure_physical.py), а не только через сервер.

# DOCX: заголовки, таблицы, свойства документа.
python-docx==1.2.0
# XLSX: электронные таблицы. Импортируется навыком legal_summarizer.
openpyxl==3.1.5
# XLS (старый двоичный формат) — openpyxl его не читает, нужен свой.
xlrd==2.0.2
# PDF: текстовый слой и страницы. Импортируется 6 модулями навыка.
pypdf==5.9.0
# PDF: таблицы и разметка страниц (отдельный слой поверх pypdf).
pdfplumber==0.11.10
# PPTX: презентации.
python-pptx==1.0.2
# Растровые изображения (встраиваются в DOCX/PPTX/XLSX).
Pillow==12.3.0
# Определение кодировки текстовых файлов перед разбором.
chardet==7.6.0


# ----------------------------------------------------------------------------
# 2. ТРАНСПОРТ К enterprise-mcp
#
# Правило: сюда попадает только то, что агент использует, чтобы обратиться
# к серверу enterprise-mcp. Сервер — отдельный процесс с отдельными
# манифестами (mcp-platform/requirements.txt и mcp-platform/pyproject.toml),
# и его duckdb/numpy/faiss-cpu/pyarrow/sqlglot/rich/anyio в этом файле нет
# и не должно появляться: общий список означал бы, что обновление платформы
# требует pip install в окружение агента.
#
# Здесь ровно один пакет, и это не совпадение: у клиента enterprise-mcp
# ровно один внешний импорт.
# ----------------------------------------------------------------------------

# Клиент stdio-сессии к процессу enterprise-mcp.
# Импортирует единственный модуль агента — lib/services/
# enterprise_mcp_client.py (from mcp import ClientSession, StdioServerParameters;
# from mcp.client.stdio import stdio_client). Раньше приходил транзитивно
# через nanobot-ai; теперь импортируется напрямую, поэтому версия
# фиксируется здесь, иначе смена каркаса молча меняла бы транспорт.
#
# Версию держит и платформа (в своих манифестах), но это разные окружения:
# у клиента и у сервера одна и та же версия mcp — условие совместимости
# протокола, а не признак общей установки.
mcp==1.27.1
